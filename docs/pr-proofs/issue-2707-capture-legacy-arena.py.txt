#!/usr/bin/env python3
"""Capture the retained LEGACY arena save (#2707) on the PRE-change build.

Run from a worktree of the base commit (5450be9f5, master before #2707):

    cabal build all
    python3 capture_legacy_arena.py --port 9291 --out /tmp/legacy_arena.synworld

Boots one headless engine, creates an arena page, expands it past the 5x5
startup footprint the only way the old build supports (the camera loader,
which supplies ONE loam cell per column, `generateFlatChunk`), raises one
tile inside the footprint and one tile in the expanded chunk (3,0), moves
the camera back (evicting and re-supplying the footprint through the same
shallow path), saves through the production path and copies the written
world.synworld to --out. The engine is quit through engine.quit().
"""
import argparse, os, shutil, sys, time
sys.path.insert(0, os.path.join(os.getcwd(), "tools"))
from probelib import boot, send, quit_engine, poll_until

ap = argparse.ArgumentParser()
ap.add_argument("--port", type=int, default=9291)
ap.add_argument("--out", required=True)
a = ap.parse_args()
PAGE, SLOT = "legacy_arena", "legacy_arena_2707"
IN_EDIT, OUT_EDIT = (6, 6), (56, 8)          # chunk (0,0) and chunk (3,0)

proc = boot(a.port, log="/tmp/legacy_arena_capture.log", label="legacy arena")
try:
    send(a.port, f"world.initArena('{PAGE}'); return 'ok'")
    send(a.port, f"world.show('{PAGE}'); return 'ok'")
    send(a.port, "return world.waitForInit(60)", timeout=70)
    def terr(x, y):
        # getTerrainAt answers (z, ...) or nil; keep the first token
        raw = send(a.port, f"return world.getTerrainAt({x}, {y})").strip()
        return raw.split()[0] if raw.split() else "nil"
    print("start terrain at edit tiles:", terr(*IN_EDIT), terr(*OUT_EDIT))
    # camera-driven expansion beyond the footprint (old explicit demand crashes)
    send(a.port, "camera.goToTile(56, 8); return 'ok'")
    if not poll_until(60, lambda: terr(*OUT_EDIT) not in ("nil", "null", "")):
        sys.exit("expanded chunk (3,0) never loaded")
    out_base = int(float(terr(*OUT_EDIT)))
    send(a.port, f"return world.addTile('{PAGE}', {OUT_EDIT[0]}, {OUT_EDIT[1]}, 1)")
    if not poll_until(30, lambda: terr(*OUT_EDIT) not in ("nil", "null", "") and int(float(terr(*OUT_EDIT))) == out_base + 1):
        sys.exit("outside-footprint edit never landed")
    # back to the origin: footprint chunks are re-supplied by the same shallow path
    send(a.port, "camera.goToTile(0, 0); return 'ok'")
    if not poll_until(60, lambda: terr(*IN_EDIT) not in ("nil", "null", "")):
        sys.exit("footprint never reloaded")
    in_base = int(float(terr(*IN_EDIT)))
    send(a.port, f"return world.addTile('{PAGE}', {IN_EDIT[0]}, {IN_EDIT[1]}, 1)")
    if not poll_until(30, lambda: int(float(terr(*IN_EDIT))) == in_base + 1):
        sys.exit("inside-footprint edit never landed")
    print(f"edits: in {IN_EDIT} {in_base}->{in_base+1}, out {OUT_EDIT} {out_base}->{out_base+1}")
    # leave the expanded chunk resident so the save is taken with it live
    send(a.port, "camera.goToTile(28, 4); return 'ok'")
    time.sleep(2)
    saved = send(a.port, f"return engine.saveWorld('{PAGE}', '{SLOT}')")
    if saved.strip() != "true":
        sys.exit(f"saveWorld returned {saved!r}")
    src = os.path.join("saves", SLOT, "world.synworld")
    if not poll_until(60, lambda: os.path.exists(src)):
        sys.exit("save never appeared")
    time.sleep(2)
    shutil.copyfile(src, a.out)
    print("captured", a.out, os.path.getsize(a.out), "bytes")
finally:
    quit_engine(a.port, proc)
    shutil.rmtree(os.path.join("saves", SLOT), ignore_errors=True)
